**Goal:** Introduce a per-note-type `NoteTypeCustomization` record and make `build_structured_prompt` config-driven, so the configurable #33 behaviors are controlled per note-type while defaults reproduce today's exact output.

**Architecture:** A new 1:1 model keyed to `DoctorNoteSchema` holds typed presets + a free-text field. The global numbered instruction block in `app/jobs/summarize.py` is split into fixed house rules + per-preset behavior modules + appended free text. `summarize.perform` loads the customization (or `None` → defaults) and threads it through `generate_clinical_note` → `build_structured_prompt`. Defaults equal current behavior, so SP1 ships with zero output change.

**Tech Stack:** Python, SQLModel/`activemodel`, Alembic (native PG enums via `alembic_postgresql_enum`), pytest, uv.

```python
# app/models/note_type_customization.py
from enum import Enum

from activemodel import BaseModel
from activemodel.mixins import SoftDeletionMixin, TimestampsMixin, TypeIDMixin
from activemodel.types import TypeIDType
from sqlmodel import Field, Relationship

from .doctor_note_schema import DoctorNoteSchema


class SubjectiveSource(str, Enum):
    patient_only = "patient_only"
    patient_and_doctor = "patient_and_doctor"


class PronounStyle(str, Enum):
    infer_from_transcript = "infer_from_transcript"
    neutral = "neutral"


class PlanOrdering(str, Enum):
    patient_then_doctor = "patient_then_doctor"
    as_discussed = "as_discussed"


class NoteTypeCustomization(
    BaseModel,
    TimestampsMixin,
    SoftDeletionMixin,
    TypeIDMixin("note_type_customization"),
    table=True,
):
    """Per-note-type overrides for the configurable note-generation behaviors.

    A missing row (or a row with all defaults) reproduces the platform's
    current note-generation behavior. Edits affect only this one
    DoctorNoteSchema — never another provider or note type.
    """

    doctor_note_schema_id: TypeIDType = DoctorNoteSchema.foreign_key(unique=True)
    doctor_note_schema: DoctorNoteSchema = Relationship()

    subjective_source: SubjectiveSource = Field(default=SubjectiveSource.patient_only)
    pronoun_style: PronounStyle = Field(default=PronounStyle.infer_from_transcript)
    plan_ordering: PlanOrdering = Field(default=PlanOrdering.patient_then_doctor)
    flag_new_medications: bool = Field(default=True)
    require_plan_reason: bool = Field(default=True)
    ros_distinct_from_subjective: bool = Field(default=True)

    custom_instructions: str | None = Field(default=None)
    "Free-text provider instructions appended verbatim to the prompt; null = none."


def default_customization() -> NoteTypeCustomization:
    """A transient (unsaved) record carrying the platform defaults.

    Used by callers that need the effective config when no row exists.
    Reading only the preset fields is safe without a schema id.
    """
    return NoteTypeCustomization()
```

Note: `DoctorNoteSchema.foreign_key(unique=True)` forwards `unique=True` to the
underlying `Field` (see `activemodel/base_model.py:170` — `field_options =
{"nullable": False} | kwargs`).

```python
    PlanOrdering,
```

```python
def _pronoun_instruction(style: PronounStyle) -> str:
    if style == PronounStyle.infer_from_transcript:
        # verbatim current instruction #8
        return (
            'Pronouns: When the transcript reveals the patient\'s pronouns -- via '
            "self-identification, the doctor's use of \"she/her\" or \"he/him\" when "
            "referring to the patient, or other unambiguous context -- the note MUST "
            "use those pronouns when referring to the patient. Do not default to "
            '"the patient" everywhere when pronouns are clear from the transcript. '
            'Use "they/them" ONLY when no pronoun usage is evident anywhere in the '
            "transcript. This overrides any clinical convention of defaulting to "
            '"the patient" -- mix in the patient\'s actual pronouns when they are '
            "known, even if the example note's style differs on this point."
        )
    return (
        'Pronouns: refer to the patient as "the patient"; do not infer pronouns from '
        "the transcript."
    )


def _subjective_instruction(source: SubjectiveSource) -> str:
    if source == SubjectiveSource.patient_only:
        # verbatim current instruction #9
        return (
            'Subjective section: source ONLY from lines spoken by the patient (lines '
            'beginning with "Patient:" in the transcript above). The doctor\'s '
            "interpretations, diagnostic impressions, follow-up questions, and "
            "clinical observations belong in the Assessment or Objective sections, "
            'never in Subjective. Phrasings like "the patient reports X" or "the '
            'patient denies Y" are acceptable as long as X and Y themselves came from '
            "a Patient: line."
        )
    return (
        "Subjective section: capture the patient's narrative. You may incorporate "
        "history elicited by the doctor's questions, framed as the patient's report "
        '(e.g. "the patient reports X"). Keep diagnostic impressions and clinical '
        "observations in the Assessment or Objective sections."
    )
```
